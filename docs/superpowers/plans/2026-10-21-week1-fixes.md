```ts
/**
 * Validated post-login return destinations.
 *
 * Only root-relative, same-site paths are allowed. Everything else —
 * external URLs, protocol-relative URLs, encoded bypasses, auth-internal
 * paths (login loops), oversized input — falls back to /account.
 */

export const DEFAULT_NEXT_PATH = "/account";

const MAX_NEXT_LENGTH = 512;

// Never allow a return destination into auth surfaces or APIs (login loops,
// callback replay, direct API hits).
const BLOCKED_PATH_PREFIXES = ["/login", "/auth", "/api"];

function decodeFully(raw: string): string {
  let candidate = raw;
  for (let i = 0; i < 3; i++) {
    let decoded: string;
    try {
      decoded = decodeURIComponent(candidate);
    } catch {
      return ""; // malformed encoding → treat as invalid
    }
    if (decoded === candidate) break;
    candidate = decoded;
  }
  return candidate;
}

export function sanitizeNextPath(raw: string | null | undefined): string {
  if (!raw) return DEFAULT_NEXT_PATH;

  const candidate = decodeFully(raw).trim();
  if (!candidate || candidate.length > MAX_NEXT_LENGTH) return DEFAULT_NEXT_PATH;

  // Must be a root-relative path.
  if (!candidate.startsWith("/")) return DEFAULT_NEXT_PATH;
  // Reject protocol-relative (//host) and backslash tricks (/\host).
  if (candidate.startsWith("//") || candidate.startsWith("/\\")) return DEFAULT_NEXT_PATH;

  // Validate the path component strictly; query/hash may follow.
  const pathPart = candidate.split(/[?#]/)[0];
  if (!/^\/[a-zA-Z0-9\-._~/]*$/.test(pathPart)) return DEFAULT_NEXT_PATH;

  const blocked = BLOCKED_PATH_PREFIXES.some(
    (prefix) => pathPart === prefix || pathPart.startsWith(`${prefix}/`)
  );
  if (blocked) return DEFAULT_NEXT_PATH;

  return candidate;
}

/**
 * Client helper: /login?next=<current path> for tool components that must
 * send the user back to the exact tool after signing in. Uses pathname only —
//  * never search params or state — so no user input leaks into the URL.
 */
export function loginUrlWithNext(): string {
  const target =
    typeof window !== "undefined" ? sanitizeNextPath(window.location.pathname) : DEFAULT_NEXT_PATH;
  return `/login?next=${encodeURIComponent(target)}`;
}
```

```ts
/**
 * Best-effort sessionStorage drafts so tool state survives a login
 * round-trip in the same tab. Values are JSON; failures (private mode,
 * quota) are silently ignored — drafts are a convenience, never a
 * source of truth. Nothing here is sent over the network or logged.
 */

const PREFIX = "goeast:draft:";

export function saveDraft<T>(key: string, value: T): void {
  if (typeof window === "undefined") return;
  try {
    window.sessionStorage.setItem(PREFIX + key, JSON.stringify(value));
  } catch {
    // ignore
  }
}

export function loadDraft<T>(key: string): T | null {
  if (typeof window === "undefined") return null;
  try {
    const raw = window.sessionStorage.getItem(PREFIX + key);
    return raw ? (JSON.parse(raw) as T) : null;
  } catch {
    return null;
  }
}

export function clearDraft(key: string): void {
  if (typeof window === "undefined") return;
  try {
    window.sessionStorage.removeItem(PREFIX + key);
  } catch {
    // ignore
  }
}
```

/**
 * The article page renders its own H1 from the frontmatter title; strip a
 * leading body H1 when it duplicates that title so pages never get two.

export function stripDuplicateTitleH1(content: string, title: string): string {
  const match = content.match(/^\s*#\s+(.+?)\s*(?:\n|$)/);
  if (!match) return content;
  const bodyH1 = match[1].trim().replace(/^["“'](.*)["”']$/, "$1");
  if (bodyH1.toLowerCase() === title.trim().toLowerCase()) {
    return content.slice(match[0].length);
  }
  return content;
}

- [ ] **Step 10.3: Implement `lib/text.ts`**

```ts
/**
 * Shorten text to at most `max` characters without cutting words or
 * sentences in half. Prefers the last complete sentence that fits within
 * the budget; otherwise cuts at a word boundary with an ellipsis. Never
 * adds content — only shortens what is already there.
 */
export function clipText(text: string, max: number): string {
  const clean = text.trim();
  if (clean.length <= max) return clean;

  const window = clean.slice(0, max);
  const sentenceEnd = Math.max(
    window.lastIndexOf(". "),
    window.lastIndexOf("! "),
    window.lastIndexOf("? "),
    window.lastIndexOf("。")
  );
  // Keep the terminator itself, drop the trailing space.
  if (sentenceEnd >= max * 0.4) {
    return window.slice(0, sentenceEnd + 1).trimEnd();
  }

  const lastSpace = window.lastIndexOf(" ");
  if (lastSpace > 0) {
    return window.slice(0, lastSpace).trimEnd() + "…";
  }
  return window.trimEnd() + "…";
}
```
